#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
Acheng Cinematic Director Engine - Quality Review & Self-Audit Linter.
Performs deterministic inspection on any generated storyboard, prompt package, or script.
Audits against the 10 Industrial Delivery Criteria:
1. Previs ID Compliance (132 Whitebox taxonomy)
2. Dual-Duration and Shot ID Syntax
3. Forbidden Buzzword Detection (Zero Tolerance)
4. H3 6-Module Schema Validation
5. Negative Prompt Hygiene (4-Vector targeted defense)
6. 5-Track Micro-Acting Anatomy (0-12s timeline)
7. 7-Layer Action Calculus & Kinetic Recoil
8. Tri-State Particles & Fluid Backbone
9. Colossal Scale Proof Chains (Axiom 6)
10. Soundscape Physics & Infrasound (Hz spec, <d> dialogue tags)
"""

import sys
import os
import re
import argparse

FORBIDDEN_BUZZWORDS = [
    "超清晰", "顶级画质", "神作", "极其震撼", "无比震撼", "威力巨大",
    "绝美光影", "无比宏伟", "华丽特效", "毁天灭地", "此处省略", "根据需要调整",
    "等等", "同上", "帅气逼真", "电影感十足", "细节拉满"
]

H3_MANDATORY_MODULES = [
    "subject_definitions",
    "summary",
    "retention_analysis",
    "detailed_description",
    "overall_soundscape",
    "Negative constraints"
]

def analyze_document(filepath):
    if not os.path.exists(filepath):
        return {"error": f"File not found: {filepath}"}
        
    with open(filepath, "r", encoding="utf-8", errors="replace") as f:
        text = f.read()
        
    results = {
        "filename": os.path.basename(filepath),
        "total_lines": len(text.splitlines()),
        "total_chars": len(text),
        "metrics": {},
        "violations": [],
        "warnings": [],
        "score": 0
    }
    
    score = 0
    
    # 1. Check Previs Whitebox Notation (0-10 pts)
    previs_matches = re.findall(r"Previs:\s*\[([a-zA-Z0-9_]+)\]", text, re.IGNORECASE)
    if not previs_matches:
        previs_matches = re.findall(r"\[Previs:\s*([a-zA-Z0-9_]+)\]", text, re.IGNORECASE)
        
    if len(previs_matches) >= 3:
        results["metrics"]["previs_tracking"] = f"PASS ({len(previs_matches)} Previs shots verified: {', '.join(previs_matches[:4])})"
        score += 10
    elif len(previs_matches) > 0:
        results["metrics"]["previs_tracking"] = f"PARTIAL ({len(previs_matches)} Previs shots found)"
        score += 5
    else:
        results["metrics"]["previs_tracking"] = "FAIL (No [Previs: xxx] tags detected)"
        results["violations"].append("Missing 132 Camera Moves Whitebox Previs IDs on shots.")

    # 2. Check Forbidden Buzzwords (0-15 pts, deduction on hit)
    buzzword_hits = []
    for line_idx, line in enumerate(text.splitlines(), start=1):
        for bw in FORBIDDEN_BUZZWORDS:
            if bw in line:
                buzzword_hits.append(f"Line {line_idx}: '{bw}' -> {line.strip()[:60]}...")
                
    if not buzzword_hits:
        results["metrics"]["buzzword_purity"] = "PASS (100% Zero-Tolerance Clean, 0 buzzwords)"
        score += 15
    else:
        results["metrics"]["buzzword_purity"] = f"FAIL ({len(buzzword_hits)} forbidden buzzwords detected)"
        for bh in buzzword_hits[:5]:
            results["violations"].append(f"Forbidden Buzzword: {bh}")
        score += max(0, 15 - len(buzzword_hits) * 3)

    # 3. Check H3 6-Module Schema (0-15 pts)
    modules_found = []
    modules_missing = []
    for mod in H3_MANDATORY_MODULES:
        if mod.lower() in text.lower():
            modules_found.append(mod)
        else:
            modules_missing.append(mod)
            
    if len(modules_found) == len(H3_MANDATORY_MODULES):
        results["metrics"]["h3_schema_compliance"] = f"PASS (All 6 H3 official modules present)"
        score += 15
    elif len(modules_found) >= 3:
        results["metrics"]["h3_schema_compliance"] = f"PARTIAL ({len(modules_found)}/6 modules: Missing {', '.join(modules_missing)})"
        score += 8
    else:
        results["metrics"]["h3_schema_compliance"] = f"NOT_APPLICABLE or INCOMPLETE ({len(modules_found)}/6 modules)"
        results["warnings"].append(f"Incomplete H3 schema: Missing {', '.join(modules_missing)}")
        score += 4

    # 4. Check Negative Prompt Hygiene (0-10 pts)
    neg_block = re.search(r"Negative constraints:(.*)", text, re.IGNORECASE)
    if neg_block:
        neg_text = neg_block.group(1)
        has_vectors = any(tag in neg_text for tag in ["[Anatomy]", "[Layout]", "[Material]", "[Screen]"]) or "no split-screen" in neg_text
        has_junk = any(junk in neg_text for junk in ["ugly", "worst quality", "bad art", "lowres"])
        if has_vectors and not has_junk:
            results["metrics"]["negative_hygiene"] = "PASS (4-Vector Targeted Defense, zero junk dumping)"
            score += 10
        elif not has_junk:
            results["metrics"]["negative_hygiene"] = "PASS (Standard targeted clean block)"
            score += 8
        else:
            results["metrics"]["negative_hygiene"] = "WARN (Contains junk words in negative prompt)"
            results["warnings"].append("Negative prompt contains generic junk words violating negative hygiene.")
            score += 5
    else:
        results["metrics"]["negative_hygiene"] = "FAIL (No Negative constraints block found)"
        results["violations"].append("Missing Negative constraints block.")

    # 5. Check 5-Track Micro-acting Physiology (0-10 pts)
    acting_markers = ["眼神", "瞳孔", "呼吸", "喉结", "下颌", "肩颈", "肌肉", "五指", "抽搐"]
    acting_hits = [m for m in acting_markers if m in text]
    has_timing = bool(re.search(r"\d+-\d+s|\d+\.\d+s", text))
    if len(acting_hits) >= 4 and has_timing:
        results["metrics"]["micro_acting_depth"] = f"PASS (Multi-track physiology verified: {', '.join(acting_hits[:4])})"
        score += 10
    elif len(acting_hits) >= 2:
        results["metrics"]["micro_acting_depth"] = f"PARTIAL (Found {len(acting_hits)} physiological cues)"
        score += 6
    else:
        results["metrics"]["micro_acting_depth"] = "WARN (Low physical micro-acting markers)"
        score += 3

    # 6. Check Action Kinetic Calculus & Impact Freeze (0-10 pts)
    kinetic_markers = ["受力", "发力", "反作用", "冲击波", "定格", "阻抗", "撞击", "激波", "碎裂", "齿轮", "液压"]
    kinetic_hits = [m for m in kinetic_markers if m in text]
    if len(kinetic_hits) >= 4:
        results["metrics"]["action_kinetic_calculus"] = f"PASS (Rigorous kinetics verified: {', '.join(kinetic_hits[:5])})"
        score += 10
    else:
        results["metrics"]["action_kinetic_calculus"] = f"WARN (Only {len(kinetic_hits)} kinetic markers)"
        score += 5

    # 7. Check Tri-State Particles & Fluid Backbone (0-10 pts)
    fluid_check = any(w in text for w in ["流体", "流体主干", "射流", "光刃", "真气流"])
    particles_check = any(w in text for w in ["粒子", "火花", "火星", "碎块", "烟雾", "水汽", "激波"])
    if fluid_check and particles_check:
        results["metrics"]["vfx_fluid_particle_synergy"] = "PASS (Fluid backbone + multi-state particles verified)"
        score += 10
    elif fluid_check or particles_check:
        results["metrics"]["vfx_fluid_particle_synergy"] = "PARTIAL (Missing either fluid backbone or particle layering)"
        score += 6
    else:
        results["metrics"]["vfx_fluid_particle_synergy"] = "N/A or LOW (No high-energy VFX expressions detected)"
        score += 4

    # 8. Check Colossal Scale Proof Chains (0-10 pts)
    colossal_markers = ["米", "千米", "百米", "基准", "阴影", "消光", "云海", "破坏", "地形"]
    colossal_hits = [m for m in colossal_markers if m in text]
    if len(colossal_hits) >= 4:
        results["metrics"]["colossal_scale_proof"] = f"PASS (Multi-evidence physical scale verified: {', '.join(colossal_hits[:4])})"
        score += 10
    else:
        results["metrics"]["colossal_scale_proof"] = "NORMAL (Standard scale scene)"
        score += 7

    # 9. Check Soundscape Physics & Infrasound (0-10 pts)
    has_hz = bool(re.search(r"\d+Hz", text, re.IGNORECASE))
    has_dialogue_tag = "<d>" in text and "</d>" in text
    has_foley = any(w in text for w in ["拟音", "foley", "啸叫", "轰鸣", "金属", "踏碎", "撕裂"])
    if has_hz and (has_dialogue_tag or has_foley):
        results["metrics"]["acoustic_physics"] = f"PASS (Infrasound Hz specified + authentic foley audio)"
        score += 10
    elif has_hz or has_foley:
        results["metrics"]["acoustic_physics"] = "PARTIAL (Audio specified, missing exact Hz or dialogue tags)"
        score += 6
    else:
        results["metrics"]["acoustic_physics"] = "WARN (Vague sound descriptions)"
        score += 2

    # 10. Check Continuity Mutation Status (0-10 pts)
    has_audit = any(w in text for w in ["audit", "continuity", "账本", "持久化", "伤势", "战损", "cold_archive"])
    if has_audit:
        results["metrics"]["continuity_persistence"] = "PASS (Continuity ledger or mutation hooks verified)"
        score += 10
    else:
        results["metrics"]["continuity_persistence"] = "WARN (No continuity state mutation block detected)"
        score += 5

    results["score"] = score
    return results

def print_audit_report(results):
    print("=" * 80)
    print(f" ACHENG DIRECTOR ENGINE - PRODUCTION QUALITY REVIEW AUDIT")
    print(f" File: {results['filename']} | Total Lines: {results['total_lines']} | Chars: {results['total_chars']}")
    print("=" * 80)
    
    print("\n--- 10 INDUSTRIAL DELIVERY CRITERIA BREAKDOWN ---")
    for k, v in results["metrics"].items():
        status = "PASS" if "PASS" in v else ("WARN" if "WARN" in v else ("PARTIAL" if "PARTIAL" in v else "FAIL"))
        print(f"[{status:7}] {k:<30} : {v}")
        
    print("\n" + "-" * 80)
    print(f" OVERALL INDUSTRIAL COMPLIANCE SCORE: {results['score']} / 100")
    print("-" * 80)
    
    if results["violations"]:
        print("\n>>> CRITICAL DEFECTS (Must Fix):")
        for viol in results["violations"]:
            print(f"  [X] {viol}")
            
    if results["warnings"]:
        print("\n>>> OPTIMIZATION SUGGESTIONS (Should Refine):")
        for warn in results["warnings"]:
            print(f"  [!] {warn}")
            
    if results["score"] >= 85 and not results["violations"]:
        print("\n>>> VERDICT: [ACCEPTED FOR PRODUCTION] - Meets Hollywood Director Grade. <<<\n")
        return True
    else:
        print("\n>>> VERDICT: [REJECTED] - Needs refinement before production delivery. <<<\n")
        return False

def main():
    parser = argparse.ArgumentParser(description="Acheng Director Quality Review Linter")
    parser.add_argument("target", help="Path to Markdown storyboard, script, or H3 prompt file")
    args = parser.parse_args()
    
    results = analyze_document(args.target)
    if "error" in results:
        print(f"Error: {results['error']}")
        sys.exit(1)
        
    passed = print_audit_report(results)
    sys.exit(0 if passed else 1)

if __name__ == "__main__":
    main()
