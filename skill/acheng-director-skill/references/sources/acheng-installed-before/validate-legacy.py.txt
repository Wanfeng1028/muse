def log(msg, success=True):
    prefix = "[PASS]" if success else "[FAIL]"
    print(f"{prefix} {msg}")

def validate_yaml_schemas(base_dir):
    print("\n--- Validating YAML Schemas ---")
    all_ok = True
    for root, _, files in os.walk(base_dir):
        for f in files:
            if f.endswith(".yaml") or f.endswith(".yml"):
                fpath = os.path.join(root, f)
                rel = os.path.relpath(fpath, base_dir)
                try:
                    with open(fpath, "r", encoding="utf-8") as fp:
                        content = fp.read()
                        yaml.safe_load(content)
                    log(f"Valid YAML: {rel}")
                except Exception as e:
                    log(f"Corrupt YAML {rel}: {e}", success=False)
                    all_ok = False
    return all_ok

def validate_deep_inspection_rules(base_dir):
    print("\n--- Validating Deep Architecture & Anti-Laziness Axioms ---")
    all_ok = True
    for rel_path, marker, label in DEEP_INSPECTION_RULES:
        fpath = os.path.join(base_dir, rel_path)
        if not os.path.exists(fpath):
            log(f"File Missing for {label}: {rel_path}", success=False)
            all_ok = False
            continue
        with open(fpath, "r", encoding="utf-8") as fp:
            text = fp.read()
            if marker in text:
                log(f"Verified {label} present in {rel_path}")
            else:
                log(f"Missing {label} in {rel_path}", success=False)
                all_ok = False
    return all_ok

def sync_to_global_mirror(source_dir, target_dir, label):
    print(f"\n--- Synchronizing Desktop Project to {label} ---")
    os.makedirs(target_dir, exist_ok=True)
    for root, dirs, files in os.walk(source_dir):
        rel = os.path.relpath(root, source_dir)
        dest_root = os.path.join(target_dir, rel) if rel != "." else target_dir
        os.makedirs(dest_root, exist_ok=True)
        for f in files:
            src_f = os.path.join(root, f)
            dst_f = os.path.join(dest_root, f)
            shutil.copy2(src_f, dst_f)
    log(f"Successfully mirrored all files to {target_dir}")

def validate_mirror_parity(source_dir, target_dir, label):
    print(f"\n--- Validating Desktop vs {label} Mirror Parity ---")
    all_ok = True
    for root, _, files in os.walk(source_dir):
        for f in files:
            rel = os.path.relpath(os.path.join(root, f), source_dir)
            global_f = os.path.join(target_dir, rel)
            if not os.path.exists(global_f):
                log(f"Mirror Missing in {label}: {rel}", success=False)
                all_ok = False
            else:
                with open(os.path.join(root, f), "rb") as f1, open(global_f, "rb") as f2:
                    if f1.read() == f2.read():
                        pass
                    else:
                        log(f"Mirror Content Mismatch in {label}: {rel}", success=False)
                        all_ok = False
    if all_ok:
        log(f"Desktop and {label} are 100% byte-for-byte identical!")
    return all_ok
